cmake_minimum_required(VERSION 3.30)
project(LexHelperIconFailureTests LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Windows / approved MSVC required")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Separate build tree; reuse production libraries without touching the root CMake file.
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(LEXHELPER_CORE_ONLY OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. lexhelper EXCLUDE_FROM_ALL)
add_executable(lexhelper_icon_failure_tests icon_failure_tests.cpp ../../resources/app.rc)
target_link_libraries(lexhelper_icon_failure_tests PRIVATE lexhelper_core lexhelper_deploy
    lexhelper_build_options comctl32 comdlg32 shell32 shlwapi ole32 uuid advapi32)
target_link_options(lexhelper_icon_failure_tests PRIVATE /MANIFEST:NO)
enable_testing()
add_test(NAME icon_failure COMMAND lexhelper_icon_failure_tests)
set_tests_properties(icon_failure PROPERTIES TIMEOUT 30)
