cmake_minimum_required(VERSION 3.30)
project(LexHelper VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "LexHelper requires Windows and the approved MSVC toolchain")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(LEXHELPER_CORE_ONLY "Build only the codec and its tests (isolated development)" OFF)
add_library(lexhelper_build_options INTERFACE)
target_compile_options(lexhelper_build_options INTERFACE /utf-8 /W4 /permissive- /Zc:__cplusplus)
target_compile_definitions(lexhelper_build_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(lexhelper_puff STATIC third_party/puff/puff.c)
target_include_directories(lexhelper_puff PUBLIC third_party/puff)
target_compile_options(lexhelper_puff PRIVATE /W3)
add_library(lexhelper_core STATIC src/core/lex.cpp src/core/text.cpp src/core/wbc.cpp src/core/zip.cpp src/core/io.cpp)
target_include_directories(lexhelper_core PUBLIC include)
target_link_libraries(lexhelper_core PRIVATE lexhelper_puff PUBLIC lexhelper_build_options)
if(BUILD_TESTING)
    add_executable(lexhelper_core_tests tests/core_tests.cpp)
    target_link_libraries(lexhelper_core_tests PRIVATE lexhelper_core)
    add_test(NAME core COMMAND lexhelper_core_tests)
endif()
if(NOT LEXHELPER_CORE_ONLY)
    enable_language(RC)
    add_library(lexhelper_deploy STATIC src/deploy/deploy.cpp)
    target_include_directories(lexhelper_deploy PUBLIC include)
    target_link_libraries(lexhelper_deploy PUBLIC lexhelper_core lexhelper_build_options PRIVATE advapi32 bcrypt shell32 ole32 uuid)
    add_executable(LexHelper WIN32 src/gui/main.cpp resources/app.rc)
    target_link_libraries(LexHelper PRIVATE lexhelper_core lexhelper_deploy lexhelper_build_options comctl32 comdlg32 shell32 shlwapi ole32 uuid advapi32)
    # GUI 的 RC 嵌入 app.manifest；禁止链接器另生成管理员 manifest。
    target_link_options(LexHelper PRIVATE /MANIFEST:NO)
    if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/deploy_tests.cpp")
        add_executable(lexhelper_deploy_tests tests/deploy_tests.cpp)
        target_link_libraries(lexhelper_deploy_tests PRIVATE lexhelper_deploy)
        add_test(NAME deploy COMMAND lexhelper_deploy_tests)
        # Test-only TU includes deploy.cpp to exercise its private call sites.
        # Do not link lexhelper_deploy (duplicate definitions / production hooks).
        add_executable(lexhelper_full_security_tests tests/full_security_tests.cpp)
        target_link_libraries(lexhelper_full_security_tests PRIVATE lexhelper_core advapi32 bcrypt shell32 ole32 uuid)
        add_test(NAME full_security COMMAND lexhelper_full_security_tests)
        add_test(NAME full_security_missing_native COMMAND lexhelper_full_security_tests --missing-native)
        add_test(NAME full_security_missing_object COMMAND lexhelper_full_security_tests --missing-object)
        # Explicit diagnostic only: never register an elevation-dependent CTest.
        add_executable(lexhelper_full_security_native_probe tests/full_security_native_probe.cpp)
        target_link_libraries(lexhelper_full_security_native_probe PRIVATE lexhelper_core advapi32 bcrypt shell32 ole32 uuid)
        # Ordinary-token parent support / internal contracts; explicit root only.
        # Never elevate this runner and do not register a root-creating CTest.
        add_executable(lexhelper_full_security_native_contract tests/full_security_native_contract_tests.cpp)
        target_link_libraries(lexhelper_full_security_native_contract PRIVATE lexhelper_core advapi32 bcrypt shell32 ole32 uuid)
    endif()
endif()
